find_package(PkgConfig REQUIRED)
pkg_check_modules(POLKIT_AGENT REQUIRED IMPORTED_TARGET polkit-agent-1)
mayonnaise_app(forma-polkit-agent ID io.github.skjsbsnq.Forma.PolkitAgent
  SOURCES main.cpp dialog.cpp backend.cpp ../app_store/json_lite.cpp)
target_link_libraries(forma-polkit-agent PRIVATE PkgConfig::POLKIT_AGENT)
install(FILES data/forma-polkit-agent.service
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/systemd/user COMPONENT forma-polkit-agent)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/forma-polkit-agent COMPONENT forma-polkit-agent)
